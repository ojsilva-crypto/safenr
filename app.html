create table documentos ( id uuid primary key default uuid_generate_v4(), empresa_id text not null default 'demo', nome text, validade date, created_at timestamptz default now() ); create table ambientes ( id uuid primary key default uuid_generate_v4(), empresa_id text not null default 'demo', nome text, nivel text, created_at timestamptz default now() );
